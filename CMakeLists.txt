cmake_minimum_required(VERSION 3.25)

# Less S - an Oberli Audio plugin, started from Oberli-Audio/plugin-template.
# Build commands: DEVELOPING.md. Rules this file follows: ship-bob versioning-and-provenance.md
# (one VERSION file, provenance header, pins, platform settings), sound-design-bob D.1.6 (the
# sanitizer builds) and the harness contract (scripts/dsp-kit/harness-contract.md).

# One VERSION file feeds the project, the plugin's version string, the About panel and the
# release; the git tag is v + VERSION.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PRODUCT_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")

# The oldest macOS the plugin runs on (10.13; arm64 cannot go below 11.0). Must be set before
# project(): after it an empty cache entry wins and the binary needs the macOS it was built on.
if (NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "Minimum macOS deployment version" FORCE)
endif()

project(LessS VERSION ${PRODUCT_VERSION} LANGUAGES C CXX)

# Provenance for the About panel and a friend's report, read from git at configure time and
# written to one generated header (a target-wide definition would rebuild every JUCE module
# on each reconfigure). file(CONFIGURE) rewrites only on change.
execute_process(COMMAND git rev-parse --short=12 HEAD
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                OUTPUT_VARIABLE BUILD_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git status --porcelain --untracked-files=no
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                OUTPUT_VARIABLE PRODUCT_GIT_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if (NOT BUILD_SHA)
    set(BUILD_SHA "unknown")
endif()
if (PRODUCT_GIT_STATUS)
    set(BUILD_DIRTY 1)
else()
    set(BUILD_DIRTY 0)
endif()
string(TIMESTAMP BUILD_DATE "%Y-%m-%dT%H:%M:%SZ" UTC)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/BuildInfo.h" CONTENT
"#pragma once
// Generated by CMakeLists.txt from VERSION and git at configure time - do not edit.
#define PRODUCT_VERSION \"@PRODUCT_VERSION@\"
#define BUILD_SHA       \"@BUILD_SHA@\"
#define BUILD_DATE      \"@BUILD_DATE@\"
#define BUILD_DIRTY     @BUILD_DIRTY@
" @ONLY)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if (APPLE AND NOT CMAKE_OSX_ARCHITECTURES)
    # Universal by default: Ableton Live 10.1 under Rosetta loads x86_64, a friend's Live 11 or
    # 12 on an M-series loads arm64. The sanitizer presets set arm64 only.
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Build architectures for macOS" FORCE)
endif()

# ---------------------------------------------------------------------------
# JUCE, pinned by commit (a tag can move): 8.0.16-security. 8.0.11 is the first JUCE whose
# bundled VST 3 SDK is MIT; buildinfo.py check refuses anything older. The pin of the family is
# in the registry (ship-bob registry/plugins.toml, family.juce); bump it there, then here.
# To build against a checkout on disk: cmake -B build -DFETCHCONTENT_SOURCE_DIR_JUCE=/path/to/JUCE
# ---------------------------------------------------------------------------
include(FetchContent)
FetchContent_Declare(
    JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG        ea77444750de45e036e0279bb0291144bf8e3688
)
FetchContent_MakeAvailable(JUCE)

# Local fixes on that commit until upstream carries them (cmake/PatchJuce.cmake): applied to
# the fetched tree, only checked on a FETCHCONTENT_SOURCE_DIR_JUCE checkout.
include(cmake/PatchJuce.cmake)
if (FETCHCONTENT_SOURCE_DIR_JUCE)
    oberli_patch_juce("${juce_SOURCE_DIR}" CHECK_ONLY)
else()
    oberli_patch_juce("${juce_SOURCE_DIR}" APPLY)
endif()

# Release symbols (ship-bob windows-packaging.md, macos-signing.md): dSYM / PDB per format,
# the shipped binary stripped on macOS. Set by the release preset and the workflow.
option(PRODUCT_RELEASE_SYMBOLS "Keep a dSYM / PDB per format and strip the shipped binary" OFF)
if (PRODUCT_RELEASE_SYMBOLS)
    if (MSVC)
        target_compile_options(LessS PUBLIC $<$<CONFIG:Release>:/Zi>)
        target_link_options(LessS PUBLIC $<$<CONFIG:Release>:/DEBUG /OPT:REF /OPT:ICF>)
    elseif (APPLE)
        target_compile_options(LessS PUBLIC $<$<CONFIG:Release>:-g>)
        foreach(format VST3 Standalone)
            if (TARGET LessS_${format})
                add_custom_command(TARGET LessS_${format} POST_BUILD
                    COMMAND dsymutil "$<TARGET_FILE:LessS_${format}>" -o "$<TARGET_BUNDLE_DIR:LessS_${format}>.dSYM"
                    COMMAND strip -x "$<TARGET_FILE:LessS_${format}>"
                    COMMENT "dSYM and strip: ${format}"
                    VERBATIM)
            endif()
        endforeach()
    endif()
endif()

# Memory and UB safety (D.1.6): PUBLIC so the runtime is linked into the loadable bundle and
# into the harness. MSVC has ASan only.
option(PRODUCT_ASAN "Build the plugin and the harness with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
if (PRODUCT_ASAN)
    if (MSVC)
        target_compile_options(LessS PUBLIC /fsanitize=address)
        target_link_options(LessS PUBLIC /INCREMENTAL:NO)
    else()
        target_compile_options(LessS PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(LessS PUBLIC -fsanitize=address,undefined)
    endif()
endif()
option(PRODUCT_TSAN "Build the plugin and the harness with ThreadSanitizer (Tier F)" OFF)
if (PRODUCT_TSAN AND NOT MSVC)
    target_compile_options(LessS PUBLIC -fsanitize=thread)
    target_link_options(LessS PUBLIC -fsanitize=thread)
endif()

# ---------------------------------------------------------------------------
# LessSHarness: the offline renderer of scripts/dsp-kit/harness-contract.md (v3).
# Links the plugin's shared code and renders without a host, a device or the editor; every
# measurement script and the test suite drive it. Built in the sanitizer configurations too.
# ---------------------------------------------------------------------------
juce_add_console_app(LessSHarness PRODUCT_NAME "LessSHarness")
target_sources(LessSHarness PRIVATE tools/harness/Main.cpp)
target_include_directories(LessSHarness PRIVATE Source "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${CMAKE_CURRENT_BINARY_DIR}/LessS_artefacts/JuceLibraryCode")
target_compile_definitions(LessSHarness PRIVATE
    JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0
    JUCE_STRICT_REFCOUNTEDPOINTER=1 ${PRODUCT_OBERLI_DEFINITIONS})
# The harness links the plugin's SharedCode target rather than compiling the sources again
# (which is what Horizon Pad does, and what the template's comment pointed at). It has to:
# since JUCE 8 the JucePlugin_* macros are PUBLIC compile definitions of that target, not a
# generated JucePluginDefines.h, so a second compile of the same sources cannot see
# JucePlugin_Name or JucePlugin_VersionString and does not build at all. Linking also saves
# the duplicate compile of every JUCE module.
target_link_libraries(LessSHarness PRIVATE
    OberliUiData ProductData
    LessS
    juce::juce_audio_utils juce::juce_dsp juce::juce_audio_formats
    juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if (PRODUCT_ASAN)
    if (MSVC)
        target_compile_options(LessSHarness PRIVATE /fsanitize=address)
        target_link_options(LessSHarness PRIVATE /INCREMENTAL:NO)
    else()
        target_compile_options(LessSHarness PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(LessSHarness PRIVATE -fsanitize=address,undefined)
    endif()
endif()
if (PRODUCT_TSAN AND NOT MSVC)
    target_compile_options(LessSHarness PRIVATE -fsanitize=thread)
    target_link_options(LessSHarness PRIVATE -fsanitize=thread)
endif()

# Real-time safety (rule 1): the harness with RealtimeSanitizer; needs LLVM 20+ (the rtsan
# preset takes the compiler from $LLVM_ROOT, set by `toolchain.py env`). processBlock runs as a
# real-time context and any allocation, lock or system call aborts with a stack trace.
option(PRODUCT_RTSAN "Build the harness with RealtimeSanitizer" OFF)
if (PRODUCT_RTSAN)
    target_compile_options(LessSHarness PRIVATE -fsanitize=realtime)
    target_link_options(LessSHarness PRIVATE -fsanitize=realtime)

# ---------------------------------------------------------------------------
# Tests (G3 and G6): the suite drives the real DSP through the harness (tools/tests/test_plugin.py,
# from the kit's test_template.py), plus the table and bank lints. Labels: fast (every build,
# under 60 s), render (goldens), lint (the CSVs). Needs python3 with numpy and scipy; without
# them nothing is registered and a plain plugin build never fails.
#   ctest --test-dir build -L fast --output-on-failure
# ---------------------------------------------------------------------------
find_package(Python3 COMPONENTS Interpreter)
if (Python3_FOUND)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy, scipy" RESULT_VARIABLE PRODUCT_PY_DEPS OUTPUT_QUIET ERROR_QUIET)
    if (PRODUCT_PY_DEPS EQUAL 0)
        enable_testing()
        set(HARNESS_ENV "HARNESS=$<TARGET_FILE:LessSHarness>" "DSPKIT=${CMAKE_CURRENT_SOURCE_DIR}/scripts/dsp-kit"
            "PRESET_ROOT=${CMAKE_CURRENT_BINARY_DIR}/test-presets" "FACTORY_PRESETS=${CMAKE_CURRENT_SOURCE_DIR}/presets")
        foreach(label fast render)
            add_test(NAME suite_${label}
                     COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/test_plugin.py" --label ${label}
                     WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
            set_tests_properties(suite_${label} PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS ${label})
        endforeach()
        add_test(NAME harness_contract
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/dsp-kit/dspkit.py" smoke "$<TARGET_FILE:LessSHarness>"
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(harness_contract PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS fast)
        add_test(NAME parameter_table
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/check_table.py"
                         --harness "$<TARGET_FILE:LessSHarness>" --kind ${PRODUCT_KIND}
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(parameter_table PROPERTIES LABELS "fast;lint")
        add_test(NAME bounce_null
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/check_bounce.py"
                         --harness "$<TARGET_FILE:LessSHarness>" --kind ${PRODUCT_KIND}
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(bounce_null PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS fast)
        # Tier P and the brief's own category list (presets.categories in docs/brief.md),
        # not the kit's instrument-flavoured default.
        #
        # Without --harness: `presetlint bank --harness` also runs its Init-equals-the-table
        # check, and that check cannot pass for a `choice` parameter whose default is neither
        # the first nor the last entry. It compares the harness's normalised echo against the
        # *plain* index, so Less S's Mode (index 1 of 3, normalised 0.5) and Speed read as
        # errors (a skills-bob item: the kit wants `fdefault / (steps - 1)` there). The two
        # properties it would have checked are verified in the suite instead, against the
        # plugin's own normalisation: Properties.test_init_preset_equals_the_table_defaults
        # and Properties.test_preset_listing_matches_the_csv.
        add_test(NAME preset_bank
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/dsp-kit/presetlint.py" bank
                         "${CMAKE_CURRENT_SOURCE_DIR}/presets.csv" "${CMAKE_CURRENT_SOURCE_DIR}/parameters.csv"
                         --tier P --track ${PRODUCT_KIND}
                         --categories Lead,Backing,Speech,Live,Instruments
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(preset_bank PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS "fast;lint")
    else()
        message(STATUS "python3 without numpy and scipy: the test suite is not registered")
    endif()
endif()
